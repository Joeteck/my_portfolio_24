            <div className="w-fit grid grid-cols-1 gap-4 sm:grid-cols-2 lg:grid-cols-3 place-content-center justify-center bg-green-700/35 mx-auto p-4 rounded-3xl">
            {([
                { variant: "info", title: "Info Modal" },
                { variant: "success", title: "Success Modal" },
                { variant: "warning", title: "Warning Modal" },
                { variant: "error", title: "Error Modal" },
                { variant: "confirm", title: "Confirmation Modal" },
                { variant: "form", title: "Form Modal" },
            ] as const).map(({ variant, title }) => (
                <Button 
                    key={variant} 
                    onClick={() => setOpenModal(variant)} 
                    className="btn btn-secondary"
                >
                    Open {title}
                </Button>
            ))}

            </div>
            
            {openModal && (
                <Modal 
                    isOpen={!!openModal} 
                    onClose={() => setOpenModal(null)}  // ✅ Use null instead of "undefined"
                    variant={openModal} 
                    title={`${openModal.charAt(0).toUpperCase() + openModal.slice(1)} Modal`}
                >
                    {openModal === "form" ? (
                        <form className="space-y-4 text-white">
                            <input type="text" placeholder="Your Name" className="w-full text-white p-2 border rounded" />
                            <input type="email" placeholder="Your Email" className="w-full text-white p-2 border rounded" />
                            <textarea placeholder="Your Message" className="w-full text-white p-2 border rounded"></textarea>
                            <Button type="submit" className="btn btn-primary">Submit</Button>
                        </form>
                    ) : openModal === "confirm" ? (
                        <div>
                            <p>Are you sure you want to proceed?</p>
                            <div className="flex space-x-4 mt-4">
                                <Button className="btn btn-primary">Yes</Button>
                                <Button className="btn bg-red-500 hover:bg-red-700" onClick={() => setOpenModal(null)}>No</Button>
                            </div>
                        </div>
                    ) : (
                        <p>This is a {openModal} modal for testing purposes.</p>
                    )}
                </Modal>
            )}
